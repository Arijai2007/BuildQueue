cmake_minimum_required(VERSION 3.16)
project(BuildQueue VERSION 1.0.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
option(BQ_ASAN "Address and undefined behavior sanitizers" OFF)
option(BQ_TSAN "Thread sanitizer (separate build)" OFF)
if(BQ_ASAN AND BQ_TSAN)
  message(FATAL_ERROR "Use separate ASan and TSan build directories")
endif()
find_package(Threads REQUIRED)
file(GLOB CORE CONFIGURE_DEPENDS src/*.cpp)
list(REMOVE_ITEM CORE ${CMAKE_CURRENT_SOURCE_DIR}/src/main.cpp ${CMAKE_CURRENT_SOURCE_DIR}/src/CLI.cpp ${CMAKE_CURRENT_SOURCE_DIR}/src/HttpServer.cpp)
add_library(buildqueue_core ${CORE})
target_include_directories(buildqueue_core PUBLIC include third_party)
target_compile_definitions(buildqueue_core PUBLIC BUILDQUEUE_ROOT="${CMAKE_CURRENT_SOURCE_DIR}")
target_compile_options(buildqueue_core PUBLIC -Wall -Wextra -Wpedantic)
target_link_libraries(buildqueue_core PUBLIC Threads::Threads)
if(BQ_ASAN)
 target_compile_options(buildqueue_core PUBLIC -fsanitize=address,undefined -fno-omit-frame-pointer)
 target_link_options(buildqueue_core PUBLIC -fsanitize=address,undefined)
endif()
if(BQ_TSAN)
 target_compile_options(buildqueue_core PUBLIC -fsanitize=thread -fno-omit-frame-pointer)
 target_link_options(buildqueue_core PUBLIC -fsanitize=thread)
endif()
add_executable(buildqueue src/main.cpp src/CLI.cpp src/HttpServer.cpp)
target_link_libraries(buildqueue PRIVATE buildqueue_core)
file(GLOB TESTS CONFIGURE_DEPENDS tests/*.cpp)
add_executable(buildqueue_tests ${TESTS})
target_link_libraries(buildqueue_tests PRIVATE buildqueue_core)
enable_testing()
add_test(NAME lab_tests COMMAND buildqueue_tests)
set_tests_properties(lab_tests PROPERTIES WORKING_DIRECTORY ${CMAKE_CURRENT_SOURCE_DIR})
